# =============================================================================
# INSTALLATION COMMANDS
# =============================================================================

# 1. Install Docker Desktop
# Download from: https://www.docker.com/products/docker-desktop/

# 2. Install Node.js (if running frontend locally)
# Download from: https://nodejs.org/ (version 20+)

# 3. Install Python (if running backend locally)
# Download from: https://python.org/ (version 3.11+)

# 4. Install Git
# Download from: https://git-scm.com/

# 5. Clone Repository
# git clone https://github.com/gokhul/poc-platform.git
# cd poc-platform

# 6. Install Frontend Dependencies (if running locally)
# cd frontend
# npm install

# 7. Install Backend Dependencies (if running locally)
# cd backend
# pip install -r requirements.txt

# 8. Start Application
# docker-compose up --build -d
